{{/* Summary Shortcode ================= Highlights a section of content as an executive summary. Usage: {{< summary >}} Your summary text goes here. {{< /summary >}} Optional parameters: title: Custom title. Defaults to the translated "executive_summary" string. icon: Bootstrap Icon class (default: bi bi-lightbulb-fill) or an image path/URL. color: Theme-aware or fixed color. Default: "primary". gradient: false, "dark-to-light", "light-to-dark", "center-dark", or "center-light". Default: "center-dark". gradient_direction: "vertical", "horizontal", or "all". Default: "horizontal". The selected color controls the summary border, title, icon box, and background. The background gradient uses the shared gradient-resolver helper. The body is Markdown and is rendered with markdownify. */}} {{- $title := .Get "title" | default (i18n "executive_summary") -}} {{- $icon := .Get "icon" | default "bi bi-lightbulb-fill" -}} {{- $color := .Get "color" | default "primary" -}} {{- $gradient := .Get "gradient" -}} {{- if eq $gradient "" -}} {{- $gradient = "center-dark" -}} {{- end -}} {{- $gradientDirection := .Get "gradient_direction" | default "horizontal" -}} {{- $isImage := findRE `(?i)\.(svg|png|webp|jpe?g|gif)(\?.*)?$` $icon -}} {{- $colorLogic := partial "helpers/color-resolver.html" (dict "color" $color) -}} {{- $gradientLogic := partial "helpers/gradient-resolver.html" (dict "color" $colorLogic.color "gradient" $gradient "gradient_direction" $gradientDirection ) -}} {{- $backgroundClass := $colorLogic.bg_class -}} {{- $backgroundStyle := "" -}} {{- if $gradientLogic.enabled -}} {{- $backgroundClass = "" -}} {{- $backgroundStyle = $gradientLogic.style -}} {{- end -}}
{{- if $isImage -}} {{- else -}} {{- end -}}
