{{/* quote Hugo shortcode for rendering a styled quotation block. Usage: {{< quote >}} This is the quoted text. {{< /quote >}} Parameters: background Theme-aware or fixed background color. Defaults to "primary". gradient Gradient style. Defaults to false. When false, the background is a solid color. Supported values follow the shared gradient resolver: false, "dark-to-light", "light-to-dark", "center-dark", "center-light". show_quote Whether to show the opening and closing quotation marks. Defaults to true. Input: Markdown or text placed between the shortcode tags. Notes: - The shortcode processes .Inner with Hugo's markdownify function. - Background and text colors are resolved through the shared color resolver. - Gradient rendering is handled by the shared gradient resolver. */}} {{- $background := .Get "background" | default "primary" -}} {{- $gradient := .Get "gradient" | default false -}} {{- if eq $gradient "" -}} {{- $gradient = false -}} {{- end -}} {{- $showQuote := .Get "show_quote" | default true -}} {{- if eq $showQuote "" -}} {{- $showQuote = true -}} {{- else -}} {{- $showQuote = $showQuote | lower | eq "true" -}} {{- end -}} {{- $colorLogic := partial "helpers/color-resolver.html" (dict "color" $background) -}} {{- $gradientLogic := partial "helpers/gradient-resolver.html" (dict "color" $background "gradient" $gradient "gradient_direction" "horizontal" ) -}}

{{- if $showQuote -}} “ {{- end -}} {{ .Inner | markdownify }} {{- if $showQuote -}} ” {{- end -}}
