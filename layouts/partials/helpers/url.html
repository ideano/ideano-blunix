{{- $url := .url | default "" -}} {{- $result := "" -}} {{- if not $url -}} {{- $result = "" -}} {{- else if or (hasPrefix $url "http://") (hasPrefix $url "https://") (hasPrefix $url "//") (hasPrefix $url "mailto:") (hasPrefix $url "tel:") (hasPrefix $url "#") -}} {{- $result = ($url | safeURL) -}} {{- else -}} {{- $result = (relref .Root $url) -}} {{- end -}} {{- return $result -}}
