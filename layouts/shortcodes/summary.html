The body is Markdown and is rendered with markdownify. */}} {{- $title := .Get "title" | default (i18n "executive_summary") -}} {{- $icon := .Get "icon" | default "bi bi-lightbulb-fill" -}} {{- $isImage := findRE `(?i)\.(svg|png|webp|jpe?g|gif)(\?.*)?$` $icon -}}
