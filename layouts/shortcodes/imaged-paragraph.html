{{/* imaged-paragraph Parameters: src - image source (required) alt - image alt text title - optional title displayed centered at the bottom of the image side - left or right; defaults to left width - image width; defaults to 35% rounded - optional border radius: none, sm, md, lg, xl, 2xl, 3xl, full class - additional image classes sizes - responsive image sizes loading - image loading mode; defaults to lazy quality - responsive image quality; defaults to 85 Example: {{< imaged-paragraph src="/images/example.webp" alt="Example image" title="Example image title" side="left" width="35%" rounded="xl" >}} This is the paragraph text beside the image. {{< /imaged-paragraph >}} */}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $title := .Get "title" | default "" -}} {{- $side := lower (.Get "side" | default "left") -}} {{- $width := .Get "width" | default "35%" -}} {{- $rounded := lower (.Get "rounded" | default "none") -}} {{- $loading := .Get "loading" | default "lazy" -}} {{- $sizes := .Get "sizes" | default "100vw" -}} {{- $quality := .Get "quality" | default 85 -}} {{- $class := .Get "class" | default "" -}} {{- if not $src -}} {{- errorf "imaged-paragraph shortcode requires a src parameter" -}} {{- end -}} {{- if not (in (slice "left" "right") $side) -}} {{- errorf "imaged-paragraph: side must be either left or right, got %q" $side -}} {{- end -}} {{- if not (in (slice "none" "sm" "md" "lg" "xl" "2xl" "3xl" "full") $rounded) -}} {{- errorf "imaged-paragraph: rounded must be one of none, sm, md, lg, xl, 2xl, 3xl, full; got %q" $rounded -}} {{- end -}} {{- $roundedClass := "" -}} {{- if eq $rounded "sm" -}} {{- $roundedClass = "rounded-sm" -}} {{- else if eq $rounded "md" -}} {{- $roundedClass = "rounded-md" -}} {{- else if eq $rounded "lg" -}} {{- $roundedClass = "rounded-lg" -}} {{- else if eq $rounded "xl" -}} {{- $roundedClass = "rounded-xl" -}} {{- else if eq $rounded "2xl" -}} {{- $roundedClass = "rounded-2xl" -}} {{- else if eq $rounded "3xl" -}} {{- $roundedClass = "rounded-3xl" -}} {{- else if eq $rounded "full" -}} {{- $roundedClass = "rounded-full" -}} {{- end -}}
{{ partial "components/image-responsive.html" (dict "src" $src "alt" $alt "class" (printf "w-full h-auto %s %s" $roundedClass $class) "loading" $loading "sizes" $sizes "quality" $quality ) }} {{- if $title -}}
{{ $title }}
